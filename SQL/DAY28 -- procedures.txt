use d19r;

select * from emp;

delimiter $$
create procedure kowshik()
begin
select * from emp;
end $$
delimiter ;

call kowshik();

delimiter $$
create procedure age()
begin
select * from emp where age>30;
end $$
delimiter ;

call age();

delimiter $$
create procedure kalyan(a int)
begin
select * from emp where age>a;
end $$
delimiter ;

select * from emp;

call kalyan(20);

call kalyan(35);

call kalyan(32);

delimiter $
create procedure dept(a varchar(100))

select * from emp where department=a;
