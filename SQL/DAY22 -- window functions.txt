window function :-- will perform operation on any type of data i.e 				numbers,strings,date
row_number()    :-- will assign a number to row from 1 to n no.of rows 

>>>select *,row_number() over() from table_name;


rank()          :-- will skip sequence numbers when duplicates are occure

>>>select *,rank() over(ordered column) from table_name;

dense_rank()    :-- will not skip the sequence number including duplicates 		    are occure

>>>select *,dense_rank() over(ordered column) from table_name;

name varchar(100),
marks int,
age int
);

select *,row_number() over(partition by name,marks,age) as sno from window_fun
