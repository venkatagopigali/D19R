extra operators :

like 
between

union       :-- will return all values from both table will return 		only once which are common
>>>select * from table1
union
select * from table2;

union all   :-- will return all values from both tables with 			duplicates

>>>select * from table1
union all
select * from table2;


intersect   :-- will return common values from both tables

>>>select * from table1
intersect
select * from table2;

except     :-- will return all values from first but with out which 		are common values with second

>>>select * from table1
except
select * from table2;
