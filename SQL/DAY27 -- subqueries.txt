use d19r;

select * from emp;

select * from emp where salary>
(select avg(salary) from emp);

select * from emp as e where salary>
(select avg(salary) from emp where department=e.department);

select department,avg(salary) from emp group by department;

select * from emp as e where salary =
(select max(salary) from emp where department=e.department);

select * from emp;

select * from emp where (city,department)=
(select city,department from emp where name='divya') and 
salary<(select salary from emp where name='divya');
